\defbibenvironment{bibliography}
  {\list
     {\mkbibbrackets{\printfield{labelnumber}}}
     {\setlength{\labelwidth}{2em}%
      \setlength{\leftmargin}{\labelwidth}%
      \setlength{\labelsep}{\biblabelsep}%
      \addtolength{\leftmargin}{\labelsep}%
      \setlength{\itemsep}{\bibitemsep}%
      \setlength{\parsep}{\bibparsep}}%
      \renewcommand*{\makelabel}[1]{\hss##1}%
      \sloppy}
  {\endlist}
  {\item}

% ========== CUSTOM DRIVER FOR @online ==========
% Format: [N] <title (italic)>. <url>. [<urldate>]
\DeclareFieldFormat[online]{title}{\emph{#1}}
\DeclareBibliographyDriver{online}{%
  \usebibmacro{bibindex}%
  \usebibmacro{begentry}%
  % Tytuł (pochylony)
  \iffieldundef{title}
    {}
    {\printfield{title}\adddot\addspace}%
  % URL z możliwością łamania
  \iffieldundef{url}
    {}
    {\url{\thefield{url}}\adddot\addspace}%
  % Data dostępu w nawiasach kwadratowych (format DD.MM.YYYY)
  \iffieldundef{urlyear}
    {}
    {\mkbibbrackets{\iffieldundef{urlday}{}{\thefield{urlday}.}\iffieldundef{urlmonth}{}{\thefield{urlmonth}.}\thefield{urlyear}}}%
  \usebibmacro{finentry}}
